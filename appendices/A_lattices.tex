\section{Lattice computations}
\label{app:lattices}

The seven integral lattices are recorded here in one place: the
hyperbolic plane $U=\mathrm{II}_{1,1}$, the Lorentzian
root lattice $\La_{II}^{2,1}$ of the Borcherds--Kac--Moody algebra
$\autcor$, the K3 cohomology lattice $\La_{K3}=\mathrm{II}_{3,19}$, the
lattice-polarized K3 sublattice \(\Lambda_{\mathrm{pol}}\), the Mukai lattice
$\widetilde H(S,\Z)\simeq\mathrm{II}_{4,20}$, the formal dyonic charge
lattice
\(\Gamma_X^{\mathrm{form}}
=\widetilde H(S,\Z)\oplus\widetilde H(S,\Z)\), and the
signature-$(3,2)$ lattice $\La^{3,2}$ that bridges $\Sp_4(\Z)$ to
the type-IV orthogonal group via the projective exterior-square image.
Each is given by
its Gram matrix or root data, signature, discriminant, and
automorphism group, with primary citations (Conway--Sloane
1999~\S2.5,~\S4,~\S16; Nikulin 1980 Theorem~1.1.2 and
Theorem~1.14.4; Mukai 1984; Gritsenko--Nikulin 1998
Part~II~\S2,~\S5).  Ranks, signatures, and the Cartan matrix of the
type-II simple roots are checked by the rational computations displayed
below.

The identities about $\Delta_5$, $\autcor$, the homogeneous orthogonal
character $\widetilde\nu_{\Delta_5}$, and the Pfaffian wall normal
form on $W^{(2)}(\La_{II}^{2,1})$ use the lattice data fixed here.
The geometric orientation character $\epsilon_o$ is compared with
this extended character as in
Theorem~\ref{thm:ch6-orientation-character}.

\begin{convention}[Sign of the bilinear form; orientation of the
time-like cone]
\label{conv:lattice-signs}
The standard mathematical convention for the hyperbolic plane $U$ is
$U=(\Z^2,((0,1),(1,0)))$, even unimodular of signature $(1,1)$ and
discriminant $-1$ (Conway--Sloane 1999~\S2.5).  This convention is
primary.

The exterior-square model of $\Sp_4(\Z)$ in
\S\ref{ssec:exceptional-iso} employs the negated form
\[
\HypPlan=\Big(\Z^2,\begin{pmatrix}0&-1\\-1&0\end{pmatrix}\Big),
\]
The time-like cone $\Cone(\La^{2,1})_+\subset
\La^{2,1}\otimes\R$ has negative square $(x,x)<0$.  The relation is $\HypPlan=U(-1)$, and $f\mapsto-f$ identifies this
hyperbolic plane with $U$.  The cohomological lattices use $U$.
The root and exterior-square lattices use the displayed BKM bases.
Negating an entire lattice form is a separate operation: it exchanges
the two signature entries and multiplies the determinant by
$(-1)^{\operatorname{rk}L}$.
\end{convention}

\subsection{The hyperbolic plane $U=\mathrm{II}_{1,1}$}
\label{app:lattice-U}

The hyperbolic plane is the unique even unimodular indefinite lattice
of signature $(1,1)$ (Conway--Sloane 1999~\S2.5; Nikulin 1980
Theorem~1.1.2).

\begin{definition}[$U=\mathrm{II}_{1,1}$]
\label{def:hyperbolic-plane}
\textnormal{[definitional]}
Let $U:=\Z e\oplus\Z f$ with Gram matrix
\[
G_U=\begin{pmatrix}0&1\\1&0\end{pmatrix},
\qquad e^2=f^2=0,\quad e\cdot f=1.
\]
\end{definition}

\begin{verification}[Rank, signature, discriminant of $U$]
\label{ver:U-invariants}
\textnormal{[computed]}
$\rk U=2$.  The eigenvalues of $G_U$ are $\pm1$, so the signature is
$(1,1)$.  The discriminant is
\[
\operatorname{disc} U=\det G_U=-1.
\]
$U$ is even (every $G_U$-diagonal entry is $0\equiv 0\pmod2$) and
unimodular ($|\det G_U|=1$).  These are the defining invariants of
$\mathrm{II}_{1,1}$ in the Conway--Sloane classification (Conway--Sloane
1999~\S2.5, Theorem~2.5.1).
\end{verification}

\begin{proposition}[$\mathrm{Aut}(U)$]
\label{prop:U-aut}
\textnormal{[proved]}
The orthogonal group of $U$ is
\[
\mathrm{O}(U)=\{\pm\Id\}\times\langle\sigma\rangle
\simeq(\Z/2)^{2},
\qquad \sigma:e\leftrightarrow f.
\]
The subgroup preserving the positive-cone component containing
$e+f$ is $\mathrm{O}^+(U)=\{\Id,\sigma\}\cong\Z/2$.
\end{proposition}

\begin{proof}
An isometry $\phi\in\mathrm{O}(U)$ preserves the isotropic vectors of
$U$.  In rank $2$ over $\Z$, the isotropic primitive vectors are
exactly $\pm e,\pm f$, so $\phi$ permutes these four vectors.  The
permutations preserving the bilinear form give the four-element group
above (Conway--Sloane 1999~\S4.1).
\end{proof}

\begin{remark}[Standard sign convention versus BKM convention]
\label{rem:U-vs-HypPlan}
The BKM convention uses
$\HypPlan=U(-1)$
in the cusp polarization of \S\ref{ssec:exceptional-iso} and in the
exterior-square model of the automorphic group, where the time-like
cone of $\La^{2,1}$ has negative square.
The map
$U\to U(-1)$, $(e,f)\mapsto(e,f)$
is a bijection of underlying $\Z$-modules; only the bilinear form
flips sign.  The lattices $\La^{2,1}$ and $\La^{3,2}$ are presented in
BKM coordinates; even-unimodular ambient lattices
($\La_{K3}$, $\widetilde H(S,\Z)$) are presented in the standard
Conway--Sloane convention.
\end{remark}

\subsection{The Lorentzian root lattice $\La_{II}^{2,1}$}
\label{app:lattice-II21}

The Borcherds--Kac--Moody algebra $\autcor$ of
\cite[Theorem~2.1, Section~5.1.1, case $(t=1,II,\overline 0)$]{GNII}
is graded by an integral lattice of signature $(2,1)$.  This is the
lattice on which the type-II simple-root Cartan matrix is
$2(2\Id_3-\J_3)$ where $\J_3$ is the all-ones matrix.

\begin{definition}[Ambient $\La^{2,1}=U(-1)\oplus\langle 2\rangle$]
\label{def:lambda-21-ambient}
\textnormal{[definitional]}
Set
\[
\La^{2,1}:=U(-1)\oplus\langle 2\rangle
=\Z e\oplus\Z f\oplus\Z h,
\qquad e\cdot f=-1,\ e^2=f^2=0,\ h^2=2.
\]
The Gram matrix in the basis $(e,f,h)$ is
\[
G_{2,1}=\begin{pmatrix}0&-1&0\\ -1&0&0\\ 0&0&2\end{pmatrix},
\qquad \det G_{2,1}=-2,\qquad
\operatorname{disc}\La^{2,1}=-2.
\]
The BKM lattice $\La^{2,1}=\HypPlan\oplus[2]$ in
$(f_2,f_3,f_{-2})$-coordinates uses exactly this Gram, with
$\HypPlan=U(-1)$ on the $(f_2,f_{-2})$-block and $[2]=\langle 2\rangle$
on the $f_3$-axis.
\end{definition}

\begin{definition}[$\La_{II}^{2,1}$ as the index-four type-II sublattice of $\La^{2,1}$]
\label{def:lambda-II-21}
\textnormal{[definitional]}
The type-II sublattice
\[
\La_{II}^{2,1}\;:=\;
\{m f_2+l f_3+n f_{-2}\in\La^{2,1}\mid m\equiv 0\,(2),\ n\equiv 0\,(2)\}
\;\subset\;\La^{2,1}
\]
is the index-$4$ sublattice on which the type-II Cartan matrix
$2(2\Id_3-\J_3)$ is the Gram of the simple roots
$(\delta_1,\delta_2,\delta_3)$ of
Proposition~\ref{prop:type-II-cartan}.  Abstractly it is the lattice
\[
\La_{II}^{2,1}\;\simeq\;U(4)\oplus\langle 2\rangle,\qquad
\operatorname{disc}\La_{II}^{2,1}=-32,
\]
matching the terminology of Chapter~\ref{ch:pentadic} and the
type-II sublattice convention of Chapter~\ref{ch:view2-bkm}.
\end{definition}

\begin{verification}[Rank, signature, discriminant of $\La_{II}^{2,1}$]
\label{ver:II21-invariants}
\textnormal{[computed]}
$\rk\La_{II}^{2,1}=3$.  The Gram of $\La_{II}^{2,1}$ in
$(\delta_1,\delta_2,\delta_3)$ of
Proposition~\ref{prop:type-II-cartan} is the Cartan matrix
$2(2\Id_3-\J_3)$ of equation~\eqref{eq:bkm-cartan-matrix}; its
determinant is
\[
\det\bigl(2(2\Id_3-\J_3)\bigr)=-32.
\]
The hyperbolic-plus-rank-one block of the ambient $\La^{2,1}$ has
signature $(2,1)$; the index-$4$ sublattice $\La_{II}^{2,1}$ inherits
this signature, so
\[
\sigma(\La_{II}^{2,1})=(2,1).
\]
The discriminant group $(\La_{II}^{2,1})^*/\La_{II}^{2,1}$ has order
$32$ as required.  The lattice $\La_{II}^{2,1}$ is even because every
$\delta_i^2=2$ and the off-diagonal couplings $(\delta_i,\delta_j)=-2$
are integral.  Changing the sign of one hyperbolic basis vector gives
the standard hyperbolic-plane convention \(U=((0,1),(1,0))\) and
presents \(\La^{2,1}\simeq U\oplus\langle 2\rangle\) and
$\La_{II}^{2,1}\simeq U(4)\oplus\langle 2\rangle$; rank, signature,
discriminant group order, automorphism group, and Cartan datum are
unchanged.
\end{verification}

\begin{remark}[Identification with \(\La^{2,1}\)]
\label{rem:II21-vs-La21}
The BKM convention uses
$\La^{2,1}=\HypPlan\oplus[2]
=\Z f_2\oplus\Z f_3\oplus\Z f_{-2}$
with Gram in $(f_2,f_3,f_{-2})$
\[
\begin{pmatrix}0&0&-1\\ 0&2&0\\ -1&0&0\end{pmatrix},
\]
matching $G_{2,1}$ of Definition~\ref{def:lambda-21-ambient} up to
the basis swap $e\mapsto f_2$, $f\mapsto f_{-2}$, $h\mapsto f_3$.
The type-II sublattice $\La_{II}^{2,1}\subset\La^{2,1}$ of
Definition~\ref{def:lambda-II-21} is the index-$4$ sublattice on
which the simple roots $\delta_1=2f_2-f_3$, $\delta_2=2f_{-2}-f_3$,
$\delta_3=f_3$ of Proposition~\ref{prop:type-II-cartan} live; it is
the even Lorentzian root lattice of $\autcor$.
\end{remark}

\begin{proposition}[Type-II simple-root Cartan]
\label{prop:type-II-cartan}
\textnormal{[proved]}
In the coordinates \((f_2,f_3,f_{-2})\) fixed by the Weyl chamber, set
\[
\delta_1=2f_2-f_3,\qquad
\delta_2=2f_{-2}-f_3,\qquad
\delta_3=f_3.
\]
Then $\delta_i^2=2$ for $i=1,2,3$ and
\[
\bigl((\delta_i,\delta_j)\bigr)_{i,j=1,2,3}
=\begin{pmatrix} 2&-2&-2\\ -2&2&-2\\ -2&-2&2\end{pmatrix}
=2(2\Id_3-\J_3),
\]
where $\J_3$ is the $3\times 3$ all-ones matrix.  Each $\delta_i$ is
type-II in the sense that $(\delta_i,\La^{2,1})=2\Z$.
\end{proposition}

\begin{proof}
Direct computation in the BKM Gram gives the matrix
$2(2\Id_3-\J_3)$, equal to the displayed Cartan matrix.
The real-root presentation uses the Cartan formalism of \cite{KAC:1}.
The type-II divisibility statement follows because each $\delta_i$
pairs against $\La^{2,1}$ with even values: $(\delta_i,f_2)$,
$(\delta_i,f_3)$, $(\delta_i,f_{-2})\in 2\Z$ for each $i$.
This is the GN root datum of \cite[Section 5.1.1, case
$(1,II,\overline0)$]{GNII}.
\end{proof}

\begin{proposition}[Weyl vector $\rho$]
\label{prop:weyl-vector-II21}
\textnormal{[proved]}
The Weyl vector
\[
\rho:=\tfrac12(\delta_1+\delta_2+\delta_3)
=f_2-\tfrac12 f_3+f_{-2}\in\La^{2,1}\otimes\Q
\]
satisfies $(\rho,\delta_i)=-1$ for $i=1,2,3$ and
\[
(\rho,\rho)=-\tfrac32,\qquad \rho\in\Cone(\La^{2,1})_+.
\]
Choose the component of the negative cone containing $\rho$.
On its unit hyperboloid, the chamber
$\Poly_{II}=\{x:(x,\delta_i)\le0,\ i=1,2,3\}$ is an ideal
hyperbolic triangle and has finite area.
\end{proposition}

\begin{proof}
In the basis $(f_2,f_3,f_{-2})$ the vector is
$\rho=(1,-1/2,1)$.  Pairing it with
$\delta_1=2f_2-f_3$, $\delta_2=2f_{-2}-f_3$ and $\delta_3=f_3$
gives $-1$ in each case.  This is the Weyl-vector convention of
\cite[(2.7)]{GNII}.  Direct computation in the BKM Gram of
Definition~\ref{def:lambda-II-21} gives
\[
(\rho,\rho)
=2(f_2,f_{-2})+\tfrac14(f_3,f_3)
=2(-1)+\tfrac14(2)=-\tfrac32.
\]

The three isotropic rays generated by
$\delta_1+\delta_2$, $\delta_2+\delta_3$ and
$\delta_1+\delta_3$ lie in the boundary of the chosen cone:
their pairing with $\rho$ is $-2$.
Each is perpendicular to its two indicated roots and has pairing
$-4$ with the third.  They are therefore the three ideal vertices
of the chamber.  Its interior contains the normalized $\rho$.
An ideal triangle has area $\pi$ for curvature $-1$.
\end{proof}

The basis
$\delta_1+\delta_3,\delta_2+\delta_3,\delta_3$
has Gram matrix $U(-4)\oplus\langle2\rangle$, with determinant
$-32$.  Since $(\rho,\delta_i)=-1$ for each generator,
$\rho\in(\La_{II}^{2,1})^\vee$.  These are arithmetic properties
of the root lattice.

\begin{proposition}[Reflection group; Nikulin's theorem]
\label{prop:nikulin-II21-reflection}
\textnormal{[proved; Nikulin 1980 Theorem~1.4.1]}
Let $W^{(2)}(\La^{2,1})$ be the subgroup of $\Orth(\La^{2,1})$
generated by reflections in primitive roots of square $2$.  Then
\[
\Orth(\La^{2,1})_+=W^{(2)}(\La^{2,1}),
\]
where the subscript $+$ denotes the index-two subgroup preserving the
time-like cone $\Cone(\La^{2,1})_+$.  Equivalently, every isometry of
$\La^{2,1}$ preserving the chosen positive cone is a product of
square-$2$ reflections.
\end{proposition}

\begin{proof}
Nikulin~\cite[Theorem~1.4.1]{NIKULIN} establishes for the rank-three
hyperbolic lattice $\HypPlan\oplus[2]$ the equality
$\Orth(\La^{2,1})_+=W^{(2)}(\La^{2,1})$; see also
Gritsenko--Nikulin~\cite[Proposition 1.5]{GN}.
\end{proof}

\begin{verification}[Type-II reflection orbit]
\label{ver:type-II-reflection-orbit}
\textnormal{[computed]}
For each $\delta\in\{\delta_1,\delta_2,\delta_3\}$, the reflection
$s_\delta:x\mapsto x-(x,\delta)\delta$ satisfies
$s_\delta(\delta)=-\delta$ and preserves the Cartan matrix on the
$\delta$-orbit.  Concretely, $s_{\delta_1}(\delta_1)=-\delta_1$,
$s_{\delta_1}(\delta_2)=\delta_2+2\delta_1$, etc.; the bilinear form
on $\{\delta_1,\delta_2,\delta_3\}$ is reflection-invariant by direct
matrix computation.
\end{verification}

\begin{remark}[The two type-$2$ divisibility classes]
\label{rem:type-I-type-II}
Each $\delta\in\La^{2,1}$ with $\delta^2=2$ has divisibility
$(\delta,\La^{2,1})\in\{\Z,2\Z\}$.  The two cases are recorded as
\emph{type I} ($(\delta,\La^{2,1})=\Z$) and \emph{type II}
($(\delta,\La^{2,1})=2\Z$); both classes generate index-finite normal
subgroups of $W^{(2)}(\La^{2,1})$:
\[
[W^{(2)}(\La^{2,1}):W^{(2)}(\La^{2,1}_I)]=2,\qquad
[W^{(2)}(\La^{2,1}):W^{(2)}(\La^{2,1}_{II})]=6,
\]
following \cite[Sections 4--5]{GN}.  The denominator algebra
$\autcor$ uses the type-II reflection chamber.
\end{remark}

\subsection{The K3 lattice $\La_{K3}=\mathrm{II}_{3,19}$}
\label{app:lattice-K3}

The integral cohomology of a K3 surface carries the ambient lattice
for its divisors and periods.  The distinction between an orthogonal
sum over $\Q$ and one over $\Z$ is essential: discriminant gluing
usually joins the algebraic and transcendental summands.

\begin{definition}[K3 lattice]
\label{def:K3-lattice}
For a complex K3 surface $S$, its K3 lattice is
\[
\La_{K3}=H^2(S,\Z)
\cong U^{\oplus3}\oplus E_8(-1)^{\oplus2},
\]
with the cup-product pairing.  Here $E_8$ is the positive definite
root lattice and $E_8(-1)$ has the negated form.
\end{definition}

\begin{verification}[K3 invariants]
\label{ver:K3-invariants}
Each summand is even and unimodular.  Their ranks and signatures give
\[
\rk\La_{K3}=22,\qquad
\sigma(\La_{K3})=(3,19),\qquad \det\La_{K3}=-1.
\]
The classification of indefinite even unimodular lattices therefore
identifies it with $\mathrm{II}_{3,19}$
\cite[Chapter~14, \S1]{HuybrechtsK3}.
\end{verification}

For a nondegenerate integral lattice $L$, put
\[
L^\vee=\{x\in L\otimes\Q:(x,L)\subset\Z\},\qquad A_L=L^\vee/L.
\]
If $L$ is even, its discriminant quadratic form is
$q_L(\bar x)=(x,x)\bmod 2\Z$.  An anti-isometry of discriminant forms
is an isomorphism $\phi:A_L\to A_T$ with
$q_T(\phi(\bar x))=-q_L(\bar x)$.

\begin{proposition}[Primitive vectors and algebraic classes]
\label{prop:K3-aut}
The group $\Orth(\La_{K3})$ is transitive on primitive vectors of each
fixed square.  For a projective K3 surface of Picard rank $\rho$,
put $N=\NS(S)$ and $T=N^\perp$.  These are primitive sublattices, with
\[
\sigma(N)=(1,\rho-1),\qquad \sigma(T)=(2,20-\rho),
\qquad
\La_{K3}\otimes\Q=(N\otimes\Q)\oplus(T\otimes\Q).
\]
There is an anti-isometry $\phi:A_N\to A_T$ for which
\[
\La_{K3}/(N\oplus T)
=\{(\bar x,\phi(\bar x)):\bar x\in A_N\}
\subset A_N\oplus A_T.
\]
In particular, $[\La_{K3}:N\oplus T]=|\det N|$.
\end{proposition}

\begin{proof}
In a lattice containing $U\oplus U$, Eichler's criterion determines
a primitive vector's orbit by its square and discriminant class.  The
latter is trivial here \cite[Chapter~14, Corollary~1.10]{HuybrechtsK3}.

The equality $N=H^{1,1}(S)\cap H^2(S,\Z)$ proves primitivity.
An orthogonal complement in an integral lattice is also primitive.  The Hodge index theorem gives the signature of $N$, hence
that of $T$ and the rational decomposition.

Let $K=\La_{K3}/(N\oplus T)\subset A_N\oplus A_T$.  Primitivity makes
each projection of $K$ injective.  Unimodularity gives
$|K|^2=|A_N||A_T|$, so both projections are surjective.  Thus $K$ is
the graph of an isomorphism.  Evenness of $\La_{K3}$ says that this
graph is isotropic for $q_N+q_T$, which is the anti-isometry assertion.
\end{proof}

For example, if $N=\Z h$ with $h=e+df$ in a hyperbolic plane and
$d>0$, then $h^2=2d$ and the complementary line is $\Z(e-df)$.
These two lines have index $2d$ in $U$.  This gives the same index
for $N\oplus T\subset\La_{K3}$ and rules out an integral splitting
in this example.  Lattice isometries also need not be automorphisms
of a fixed surface: such an automorphism preserves its Hodge
structure and ample cone.

\subsection{Lattice polarizations and discriminant gluing}
\label{app:lattice-Lambda-S}

\begin{definition}[Lattice polarization]
\label{def:lattice-polarization}
Let $N=\Lambda_{\mathrm{pol}}$ be an even lattice of signature
$(1,\rho-1)$, where $1\leq\rho\leq20$.  An ample $N$-polarization
of a K3 surface is a primitive embedding $N\hookrightarrow\NS(S)$
whose image contains an ample class.  A marking identifies this with
an embedding $N\hookrightarrow\La_{K3}$ and places the period line
in
\[
\mathcal D_{N^\perp}
=\{[\omega]\in\mathbb P(N^\perp\otimes\C):
(\omega,\omega)=0,\ (\omega,\bar\omega)>0\}.
\]
The period map to this domain is for marked surfaces.  Forgetting
the marking requires the corresponding arithmetic quotient.
\end{definition}

\begin{verification}[Complement and gluing index]
\label{ver:lambda-S-invariants}
For a primitive embedding of $N$ into the K3 lattice, its complement
$T=N^\perp$ has rank $22-\rho$ and signature $(2,20-\rho)$.  The
argument of Proposition~\ref{prop:K3-aut} gives
\[
(A_N,q_N)\cong(A_T,-q_T),\qquad
|\det N|=|\det T|=[\La_{K3}:N\oplus T].
\]
The finite-index extension is specified by the graph of this
anti-isometry in $A_N\oplus A_T$.
\end{verification}

\begin{proposition}[Embedding by discriminant gluing]
\label{prop:nikulin-embedding}
Let $N$ be even of signature $(1,\rho-1)$.
It embeds primitively in $\La_{K3}$ precisely when an even lattice
$T$ of signature $(2,20-\rho)$ admits an anti-isometry $A_N\cong A_T$.

A sufficient condition for existence and uniqueness of the embedding
up to $\Orth(\La_{K3})$ is
\[
\rho\leq19,\qquad \ell(A_N)+2\leq22-\rho,
\]
where $\ell(A_N)$ is the minimal number of generators of $A_N$.
\end{proposition}

\begin{proof}
Necessity follows from the preceding gluing calculation.  Conversely,
given $T$ and an anti-isometry $\phi$, form
\[
L_\phi=\{(x,y)\in N^\vee\oplus T^\vee:
\bar y=\phi(\bar x)\}.
\]
The graph is isotropic, so $L_\phi$ is even and integral.  Its index
$|A_N|$ over $N\oplus T$ makes it unimodular.  Its signature is
$(3,19)$, hence $L_\phi\cong\La_{K3}$.  The graph condition also
shows $L_\phi\cap(N\otimes\Q)=N$, so the embedding is primitive.

The sufficient condition is Nikulin's indefinite embedding theorem
\cite[Theorem~1.14.4]{Nikulin1980}.  Its stated form is
\cite[Chapter~14, Theorem~1.12]{HuybrechtsK3}.  The inequalities
$1<3$ and $\rho-1<19$ ensure that the complement has both signs.
The displayed bound is the required discriminant-length bound.
\end{proof}

Thus a primitive embedding requires an actual complement lattice,
not merely a proposed discriminant group.  Once an embedding exists,
the K3 period theorem realizes $N$ as the full N\'eron--Severi
lattice of a projective K3 surface.  Choosing a positive chamber
determines which classes are ample.  The explicit embeddings and
ample classes used here were constructed in
\S\ref{k3:sec:spherical}.

\begin{remark}[Mirror lattice convention]
\label{rem:mirror-lattice-convention}
Suppose $N^\perp$ contains a hyperbolic plane $U_{\mathrm{mir}}$.
Its unimodularity gives an integral splitting
\[
N^\perp=U_{\mathrm{mir}}\oplus\check N,
\qquad \check N=U_{\mathrm{mir}}^\perp\cap N^\perp.
\]
Then $\check N$ has rank $20-\rho$ and signature $(1,19-\rho)$.
This is the lattice used for the mirror polarization.  The choice of
$U_{\mathrm{mir}}$ is part of the construction.

For $N=U\oplus\langle-2\rangle$, embed $U$ in one hyperbolic
summand of $\La_{K3}$ and the vector of square $-2$ as a root in one
$E_8(-1)$ summand.  The root complement there is $E_7(-1)$.  Removing
one of the two remaining hyperbolic planes gives
\[
\check N\cong U\oplus E_7(-1)\oplus E_8(-1),
\qquad \sigma(\check N)=(1,16).
\]
The interpretation of $\Delta_5^{-2}$ as a mirror section of the
inverse Hodge line with reduced type-II Heegner support remains a
conjectural comparison.  The lattice splitting alone does not
construct that section or the compact object of
Chapter~\ref{chap:dirac-igusa-pro-object}.
\end{remark}

\subsection{The Mukai lattice $\widetilde H(S,\Z)=\mathrm{II}_{4,20}$}
\label{app:lattice-Mukai}

\begin{definition}[Cohomological and algebraic Mukai lattices]
\label{def:mukai-lattice-appA}
The cohomological Mukai lattice and its pairing are
\[
\widetilde H(S,\Z)=H^0(S,\Z)\oplus H^2(S,\Z)\oplus H^4(S,\Z),
\qquad
\langle(r,D,s),(r',D',s')\rangle=D\cdot D'-rs'-r's.
\]
For a projective K3 surface, the algebraic sublattice is
$\widetilde H_{\mathrm{alg}}(S,\Z)=\Z\oplus\NS(S)\oplus\Z$.
\end{definition}

\begin{verification}[Mukai invariants]
\label{ver:mukai-invariants}
The two outer cohomology groups form a hyperbolic plane.  Consequently
\[
\widetilde H(S,\Z)\cong U^{\oplus4}\oplus E_8(-1)^{\oplus2},
\qquad
\rk=24,\quad \sigma=(4,20),\quad \det=1.
\]
The algebraic sublattice instead has rank $\rho+2$ and signature
$(2,\rho)$.  Its determinant has absolute value $|\det\NS(S)|$,
which need not be one.
\end{verification}

\begin{proposition}[Mukai vector and Euler form]
\label{prop:mukai-vector}
For $F\in D^b\Coh(S)$,
\[
v(F)=\operatorname{ch}(F)\sqrt{\operatorname{td}(S)}
=(\rk F,c_1(F),\rk F+\operatorname{ch}_2(F))
\in\widetilde H_{\mathrm{alg}}(S,\Z),
\]
where the last coordinate is integrated over $S$.  For any $F,G$,
\[
\chi_S(F,G)=-\langle v(F),v(G)\rangle.
\]
This identifies the numerical Grothendieck group with the algebraic
Mukai lattice.
\end{proposition}

\begin{proof}
The identities $\sqrt{\operatorname{td}(S)}=1+[\mathrm{pt}]$ and
Hirzebruch--Riemann--Roch give the formulas.  The vectors of line
bundles and point sheaves generate the algebraic lattice, whose
pairing is nondegenerate.  Thus the kernel of the Mukai vector is
exactly the Euler radical, as proved in
Proposition~\ref{k3:prop:rr}.
\end{proof}

\begin{remark}[The hyperbolic sign]
\label{rem:mukai-rank-0-sign}
On $H^2$ the Mukai pairing is the cup product.  On $H^0\oplus H^4$
its matrix in the rank and point bases is
\[
J_- =\begin{pmatrix}0&-1\\-1&0\end{pmatrix}.
\]
The integral matrix $D=\operatorname{diag}(1,-1)$ satisfies
$D^tJ_-D=\left(\begin{smallmatrix}0&1\\1&0\end{smallmatrix}\right)$.
Changing the off-diagonal sign therefore leaves the lattice
isometry class and signature unchanged.  Riemann--Roch fixes the
minus sign for the specified coordinates of $v(F)$.  For instance,
$\langle v(\mathcal O_S),v(\mathcal O_p)\rangle=-1$, whereas
$\chi_S(\mathcal O_S,\mathcal O_p)=1$.
For a nonisotropic vector $v\in\widetilde H(S,\R)$, the orthogonal
reflection is
$r_v(x)=x-2\langle x,v\rangle v/\langle v,v\rangle$.
\end{remark}

\begin{proposition}[Isometries of the Mukai lattice]
\label{prop:mukai-aut}
The group $\Orth(\widetilde H(S,\Z))$ is transitive on primitive
vectors of each fixed square.  A $\C$-linear exact derived autoequivalence of $S$
induces an integral Hodge isometry of this lattice, with $H^0$ and
$H^4$ assigned Hodge type $(1,1)$.
\end{proposition}

\begin{proof}
The first assertion follows from the same primitive-vector criterion
as Proposition~\ref{prop:K3-aut}.  A Fourier--Mukai kernel $\mathcal P$ gives the action
$\alpha\mapsto p_{2*}(p_1^*\alpha\,v(\mathcal P))$.  Integrality
of $v(\mathcal P)$, the projection formula and the inverse kernel
make this an integral isometry.  Algebraicity of its Chern character
preserves the Mukai Hodge structure
\cite[Chapter~16, Lemma~3.1 and Proposition~3.2]{HuybrechtsK3}.
\end{proof}

The vectors $v(\mathcal O_S)=(1,0,1)$ and $v(\mathcal O_p)=(0,0,1)$
have squares $-2$ and $0$, respectively.  Each represents its own
primitive lattice orbit.  Transitivity of the full orthogonal group
does not assert transitivity of derived autoequivalences on objects
or on a fixed Hodge structure.

\subsection{The formal paired charge lattice}
\label{app:lattice-formal-dyonic}

\begin{definition}[Paired charges and Gram map]
\label{def:gamma-X-form}
For $X=S\times E$, define
\[
\Gamma_X^{\mathrm{form}}
=\widetilde H(S,\Z)\oplus\widetilde H(S,\Z),\qquad c=(Q,P).
\]
Its componentwise form is
\[
\langle(Q,P),(Q',P')\rangle_\Gamma
=\langle Q,Q'\rangle+\langle P,P'\rangle,
\]
and its quadratic Gram map is
\[
\Pi_X(c)=\left(\frac{Q^2}{2},Q\cdot P,\frac{P^2}{2}\right)
\in\Gamma_{\mathrm{gram}}=\Z^3.
\]
Evenness makes the half-squares integral.  The polarization of
$\Pi_X$ is the bilinear map of
Lemma~\ref{lem:gram-additivity-defect}.
\end{definition}

\begin{verification}[Paired-lattice invariants]
\label{ver:gamma-X-form-invariants}
With the componentwise form, $\Gamma_X^{\mathrm{form}}$ is even
unimodular of rank $48$, signature $(8,40)$ and determinant $1$.
Thus it is isometric to $\mathrm{II}_{8,40}$.
\end{verification}

\begin{remark}[Three pairings on paired charges]
\label{rem:gamma-X-vs-K3E}
Write $\omega_E$ for the point class of integral one.  The K\"unneth
isomorphism identifies $c=(Q,P)$ with
\[
v_X=P\otimes1_E+Q\otimes\omega_E.
\]
Besides the componentwise form, this group carries the symmetric
cross-pairing
\[
b_{\mathrm{cross}}(c,c')=Q\cdot P'+P\cdot Q'.
\]
It is the tensor of the Mukai form with the hyperbolic form on
$H^0(E,\Z)\oplus H^2(E,\Z)$, and has signature $(24,24)$.
For actual coherent complexes, the Euler form is instead
\[
\chi_X(c,c')=Q\cdot P'-P\cdot Q'.
\]
This alternating form is derived in
\S\ref{sec:gram-map-polarization}.

The two symmetric forms
are auxiliary lattice structures.  Coherent charges lie in the
paired algebraic sublattice, and the Gram map is an additional
quadratic operation on those charges.
\end{remark}

\begin{lemma}[Two hyperbolic planes and primitive lifts]
\label{lem:two-hyperbolic-planes-appA}
Choose orthogonal hyperbolic summands $U_1,U_2$ of
$\widetilde H(S,\Z)$, with bases $e_i,f_i$ and $e_i\cdot f_i=1$.
For every $(n,l,m)\in\Z^3$, the vectors
\[
Q=e_1+nf_1,\qquad P=lf_1+e_2+mf_2
\]
span a primitive rank-two sublattice and satisfy
$\Pi_X(Q,P)=(n,l,m)$.
\end{lemma}

\begin{proof}
The pairings are $Q^2=2n$, $Q\cdot P=l$, and $P^2=2m$.
The $e_1,e_2$ rows of the two-column coordinate matrix form the
identity matrix.  Its maximal minors therefore have greatest common
divisor one, which proves primitivity.
\end{proof}

\begin{verification}[Root coordinates of a lift]
\label{ver:formal-primitive-lift-packet-appA}
In the cusp coordinates fixed for the root lattice,
\[
\alpha(\Pi_X(Q,P))=2nf_2-lf_3+2mf_{-2}.
\]
The coefficient of $e_1\wedge e_2$ in $Q\wedge P$ is one.  These
are integral lattice identities for every triple.  The chosen hyperbolic planes need not be algebraic for a fixed
surface.  A coherent representative, and then a semistable one,
require additional geometric conditions.
\end{verification}

\subsection{The signature-$(3,2)$ lattice $\La^{3,2}$ and the
$\PSp_4\simeq\SO(3,2)_+$ bridge}
\label{app:lattice-32}

The exceptional isomorphism $\PSp_4(\R)\simeq\SO_+(3,2)$ identifies
the Siegel domain with the type-IV orthogonal domain of an integral
lattice $\La^{3,2}$ of signature $(3,2)$, and it identifies
\(\Sp_4(\Z)/\{\pm\Id\}\) with the exterior-square arithmetic group
\(\Gamma_{3,2}\subset\mathrm P\Orth(\La^{3,2})_+\)
(\cite[Section~3]{GN}; Borcherds 1995~\S5).  The Igusa cusp form
$\Delta_5$ lives on the Siegel side; the Borcherds product expansion
lives on the orthogonal side.

\begin{definition}[$\La^{3,2}$ in the exterior-square model]
\label{def:lambda-32}
\textnormal{[definitional]}
Let $\La^4=\Z e_1\oplus\Z e_2\oplus\Z e_3\oplus\Z e_4$ and consider
the exterior square $\wedge^2\La^4$ with the symmetric pairing
\[
u\wedge v\cdot u'\wedge v'=
[\det(u,v,u',v')]_{e_1\wedge e_2\wedge e_3\wedge e_4}\in\Z.
\]
Then $(\wedge^2\La^4,(\,,\,))\simeq\mathrm{II}_{3,3}$ as an even
unimodular lattice of signature $(3,3)$.  Fix the primitive vector
$q_0:=e_1\wedge e_3+e_2\wedge e_4\in\wedge^2\La^4$, for which
$q_0^2=-2$, and define
\[
\La^{3,2}:=q_0^\perp\subset\wedge^2\La^4.
\]
\end{definition}

\begin{verification}[Rank, signature, discriminant of $\La^{3,2}$]
\label{ver:lambda-32-invariants}
\textnormal{[computed]}
$\rk\wedge^2\La^4=\binom{4}{2}=6$.  Taking the orthogonal complement of
the primitive vector $q_0$ gives $\rk\La^{3,2}=5$.  The
explicit Gram matrix of $\La^{3,2}$ in the basis
$(f_1,f_2,f_3,f_{-2},f_{-1})$ is
\[
G_{3,2}=\begin{pmatrix}
0&0&0&0&-1\\
0&0&0&-1&0\\
0&0&2&0&0\\
0&-1&0&0&0\\
-1&0&0&0&0
\end{pmatrix},
\]
where $f_1=e_1\wedge e_2$, $f_2=e_2\wedge e_3$,
$f_3=e_1\wedge e_3-e_2\wedge e_4$, $f_{-2}=e_4\wedge e_1$,
$f_{-1}=e_4\wedge e_3$.  Direct computation: the matrix is
antidiagonal except for the central $h^2=2$ entry, with off-diagonal
entries $-1,-1,-1,-1$.  Expansion gives
\[
\det G_{3,2}=2\cdot(-1)^4\cdot\operatorname{sgn}(1,2,3,4\mapsto4,3,2,1)
=2\cdot 1\cdot(+1)=+2,
\]
since the permutation $(1,2,3,4)\mapsto(4,3,2,1)$ has six inversions
and even parity.  Therefore
\[
\sigma(\La^{3,2})=(3,2),\qquad
\operatorname{disc}\La^{3,2}=2.
\]
This is a rational computation in the exterior-square lattice.
\end{verification}

\begin{proposition}[$\La^{3,2}\simeq U\oplus U\oplus\langle 2\rangle$
in the fixed convention; $\La^{3,2}\simeq U(-1)\oplus
U(-1)\oplus\langle 2\rangle$ in the standard convention]
\label{prop:lambda-32-decomposition}
\textnormal{[proved]}
In the BKM convention of the exterior-square model,
\[
\La^{3,2}\simeq\HypPlan\oplus\HypPlan\oplus[2],
\]
with $\HypPlan=U(-1)$ and $[2]$ the rank-$1$ lattice with Gram
$(2)$.  The Gram presentation displayed in
Verification~\ref{ver:lambda-32-invariants} arises from the
$f$-basis: $(f_1,f_{-1})$ pairs as a copy of $\HypPlan$
($(f_1,f_{-1})=-1$), $(f_2,f_{-2})$ pairs as a second copy of
$\HypPlan$ ($(f_2,f_{-2})=-1$), and $f_3$ contributes
$(f_3,f_3)=2$.
\end{proposition}

\begin{proof}
The pairings $(f_1,f_{-1})=-1$, $(f_2,f_{-2})=-1$, $(f_3,f_3)=2$ are
read off from $G_{3,2}$.  The orthogonal decomposition is direct:
$\Z f_1+\Z f_{-1}\perp\Z f_2+\Z f_{-2}\perp\Z f_3$ inside
$\La^{3,2}$ because every cross-pair vanishes in $G_{3,2}$.  The
identification with $\HypPlan\oplus\HypPlan\oplus[2]$ follows
because each rank-$2$ block has Gram $((0,-1),(-1,0))=\HypPlan$
and the rank-$1$ block has Gram $(2)=[2]$.
\end{proof}

\begin{proposition}[The projective exterior-square arithmetic group]
\label{prop:psp4-iso-so32}
\textnormal{[proved; exterior-square model of $\Sp_4(\Z)$]}
The exterior-square representation
$\wedge^2$ induces an isomorphism
\[
\wedge^2:\Sp_4(\Z)/\{\pm\Id_4\}
\xrightarrow{\sim}
\Gamma_{3,2}\subset \mathrm P\Orth(\La^{3,2})_+,
\qquad
\Gamma_{3,2}:=\wedge^2\Sp_4(\Z)/\{\pm\Id_4\}.
\]
Concretely, the integral generators
\[
g_0=\begin{pmatrix}0&\Id_2\\-\Id_2&0\end{pmatrix},\quad
g_M=\begin{pmatrix}\Id_2&M\\ 0&\Id_2\end{pmatrix},\quad
g_A=\begin{pmatrix}{}^tA^{-1}&0\\ 0&A\end{pmatrix}
\quad(A\in\GL_2(\Z))
\]
of $\Sp_4(\Z)$ map under $\wedge^2$ to integral isometries of
$\La^{3,2}$, displayed as the three explicit $5\times 5$ matrices
$\wedge^2(g_0)$, $\wedge^2(g_M)$, $\wedge^2(g_A)$ in the exterior-square model.  The image is the arithmetic group through which
the Siegel modular group acts on the type-IV model.
The Siegel space $\Hpl_2$ embeds into the type-IV upper half-space
$\Hpl^{\IV}_+\subset\Proj(\La^{3,2}\otimes\C)$ via
\[
Z=\begin{pmatrix}z_1&z_2\\ z_2&z_3\end{pmatrix}
\mapsto
{}^t(z_2^2-z_1z_3,\,z_3,\,z_2,\,z_1,\,1)z_0,
\]
and the action of $\Sp_4(\Z)$ on $\Hpl_2$ corresponds to the action
of $\Gamma_{3,2}$ on $\Hpl^{\IV}_+$.
\end{proposition}

\begin{proof}
The Pfaffian pairing on $\wedge^2\La^4$ is preserved by
$\wedge^2\SL_4$.  The subgroup fixing $q_0$ is $\Sp_4(\Z)$ by
definition of the symplectic form
$B_{q_0}(x,y)e_1\wedge e_2\wedge e_3\wedge e_4=-x\wedge y\wedge q_0$.
The kernel on $q_0^\perp=\La^{3,2}$ is $\{\pm\Id_4\}$.  The
exceptional Lie-algebra isomorphism $\mathfrak{sp}_4(\R)\simeq\mathfrak{so}(3,2)$
extends to the integral form on the displayed generators.  The
discriminant group of $\La^{3,2}$ has one non-zero element; hence the
stable and full projective components coincide in the
Gritsenko--Nikulin normalization.  This is the content of
\cite[Section~3]{GN} and the exterior-square model of the
automorphic group.
\end{proof}

\begin{remark}[Embedding $\La_{II}^{2,1}\subset\La^{3,2}$]
\label{rem:II21-in-32}
The primitive sublattice
\[
\La^{2,1}=\HypPlan\oplus[2]
=\Z f_2\oplus\Z f_3\oplus\Z f_{-2}\subset\La^{3,2}
\]
of the Weyl-chamber section is the BKM root lattice: discarding the
$(f_1,f_{-1})$-block of $\La^{3,2}\simeq\HypPlan\oplus\HypPlan\oplus
[2]$ leaves $\HypPlan\oplus[2]\simeq\La^{2,1}$.  The type-II
sublattice $\La_{II}^{2,1}=\{mf_2+lf_3+nf_{-2}\mid
m,n\equiv 0\!\pmod 2\}$ is the lattice of even integral type-II
classes, on which the simple-root Cartan
\[
\bigl((\delta_i,\delta_j)\bigr)
=\begin{pmatrix}2&-2&-2\\-2&2&-2\\-2&-2&2\end{pmatrix}
\]
of Proposition~\ref{prop:type-II-cartan} is realized via
$\delta_1=2f_2-f_3$, $\delta_2=2f_{-2}-f_3$, $\delta_3=f_3$.  Every
isometry of $\La^{2,1}$ extends to an isometry of $\La^{3,2}$ by
the identity on $(\La^{2,1})^\perp\subset\La^{3,2}$.
\end{remark}

\subsection{Cross-references and consistency check}
\label{app:lattice-cross-refs}

The lattices recorded above enter as follows.

\begin{itemize}
\item $U=\mathrm{II}_{1,1}$: every $U$-summand of
$\La^{3,2}$, $\La_{K3}$, and $\widetilde H(S,\Z)$.  The relation
$\HypPlan=U(-1)$ records the BKM negated convention.
\item $\La_{II}^{2,1}\simeq U(4)\oplus\langle 2\rangle$ (with
$U(4)$ written as \(U(-4)\) in the BKM sign convention):
the BKM root lattice on which the type-II simple-root Cartan
$2(2\Id_3-\J_3)$ and the Weyl vector
$\rho=\frac12(\delta_1+\delta_2+\delta_3)$ are realized.  The
denominator algebra
$\autcor=\mathfrak g_{\Delta_5}$ is graded by $\La_{II}^{2,1}$
(\cite[Theorem~2.1]{GNII}).
\item $\La_{K3}=\mathrm{II}_{3,19}$: K3 cohomology with
intersection pairing.  Polarized K3 sublattices \(\Lambda_{\mathrm{pol}}\) embed
primitively into $\La_{K3}$.
\item \(\Lambda_{\mathrm{pol}}\): signature $(1,\rho-1)$ for $\rho\le20$.  It is the lattice-polarization sublattice of
\(\La_{K3}\), not the full Mukai lattice.
\item $\widetilde H(S,\Z)=\mathrm{II}_{4,20}$: the cohomological
Mukai lattice containing each component $Q_Y,P_Y$ of
$v_X(\mathcal I_Y)$.  Their three pairings give the Gram map
(Theorem~\ref{thm:d6-d2-d0-dictionary-appendixC}).
\item
\(\Gamma_X^{\mathrm{form}}
=\widetilde H(S,\Z)\oplus\widetilde H(S,\Z)\): the formal
additive dyonic charge lattice of $X=S\times E$, on which the
quadratic Gram-shadow projection $\Pi_X$ and the symmetric bilinear
$2$-cocycle $B$ are defined
(Definition~\ref{def:mukai-lattice}).
\item $\La^{3,2}\simeq U(-1)\oplus U(-1)\oplus\langle 2\rangle$:
the signature-$(3,2)$ orthogonal lattice that bridges
$\PSp_4(\Z)$ to the type-IV orthogonal group via the exterior-square
representation; on this lattice the Borcherds product expansion of
$\Delta_5$ is written.
\end{itemize}

\begin{verification}[End-to-end computation check]
\label{ver:lattice-rational-check}
\textnormal{[computed]}
The lattice data satisfy, in one self-contained $\Q$-arithmetic
computation:
(i) the rank-$5$ Gram matrix $G_{3,2}$ in the
$(f_1,f_2,f_3,f_{-2},f_{-1})$-basis;
(ii) the orthogonality of the symplectic invariant
$q_0=e_1\wedge e_3+e_2\wedge e_4$ to the basis;
(iii) the type-II simple-root Cartan
$2(2\Id_3-\J_3)$;
(iv) the Weyl vector $\rho=(1,-1/2,1)$ in $(f_2,f_3,f_{-2})$
coordinates and the $(\rho,\delta_i)=-1$ relation;
(v) the type-II reflection action and Cartan invariance under each
$s_{\delta_i}$.
Thus the Gram, Cartan, and
Weyl-vector data of \S\ref{app:lattice-II21} and \S\ref{app:lattice-32}
agree in the common rational lattice.
\end{verification}

\begin{remark}[Negating a lattice form]
\label{rem:discriminant-conventions}
The lattice $\La^{3,2}=(\HypPlan)^{\oplus2}\oplus\langle2\rangle$
has signature $(3,2)$ and determinant $2$.  Negating its whole form
gives
\[
\La^{3,2}(-1)\cong U^{\oplus2}\oplus\langle-2\rangle,
\]
with signature $(2,3)$ and determinant $-2$.  Their underlying
discriminant groups are both $\Z/2$, but their discriminant
quadratic forms take values $1/2$ and $-1/2$ on the nonzero class,
respectively.  Thus changing the hyperbolic basis and negating the
whole form must be distinguished.

Similarly, $\La^{2,1}=\HypPlan\oplus\langle2\rangle$ has
determinant $-2$, while its negative has determinant $2$.
The type-II sublattice has index four, hence determinant $-32$
in the chosen BKM convention.  Its discriminant group has order
$32$, whereas the ambient lattice has discriminant group of order
two.  Divisibility of a primitive ambient vector is controlled by
the latter group.  The sign of the determinant does not change this
underlying group.
\end{remark}
